\newcommand{\gear}[1]{%
  \begin{scope}[scale=#1]

    % Corps de l'engrenage
    \fill[ensaiblue!40]
      (0,0) circle (1cm);

    % Dents
    \foreach \i in {0,...,11} {
      \begin{scope}[rotate=\i*30]
        \fill[ensaiblue!30]
          (-0.17cm,0.78cm)
          rectangle
          (0.17cm,1.20cm);
      \end{scope}
    }

    % Trou central
    \fill[white]
      (0,0) circle (0.32cm);

    % Contours
    \draw[
      ensaiblue!60,
      line width=0.8pt
    ]
      (0,0) circle (1cm);

    \draw[
      darkgray!40,
      line width=0.8pt
    ]
      (0,0) circle (0.32cm);

  \end{scope}%
}

    ([xshift=-1.50cm,yshift=1.45cm]current page.south east);

    font=\small,
    text=darkgray
] at ([yshift=0.10cm]C)
    {\thepage};

\newcommand{\coverbackground}{%
\begin{tikzpicture}[remember picture,overlay]

    % ========================================================
    % TRIANGLE BLEU EN HAUT À GAUCHE
    % ========================================================
    %
    % Beaucoup plus court que précédemment.
    %

    \fill[ensailightblue]
        (current page.north west)
        --
        ([xshift=3.8cm]current page.north west)
        --
        ([yshift=-7.2cm]current page.north west)
        --
        cycle;

    % ========================================================
    % PETIT RÉSEAU DE POINTS EN HAUT
    % ========================================================

    % lignes horizontales
    \foreach \y in {-0.7,-1.7,-2.7} {

        \draw[
            ensaiblue!12,
            line width=0.5pt
        ]
        ([xshift=3.2cm,yshift=\y cm]current page.north west)
        --
        ([xshift=8.7cm,yshift=\y cm]current page.north west);
    }

    % lignes verticales
    \foreach \x in {3.5,4.5,5.5,6.5,7.5} {

        \draw[
            ensaiblue!12,
            line width=0.5pt
        ]
        ([xshift=\x cm,yshift=-0.2cm]current page.north west)
        --
        ([xshift=\x cm,yshift=-3.2cm]current page.north west);
    }

    % points
    \foreach \x in {3.5,4.5,5.5,6.5,7.5} {
        \foreach \y in {-0.7,-1.7,-2.7} {

            \fill[
                ensaiblue!25
